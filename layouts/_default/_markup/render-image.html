{{/* Markdown の画像レンダリングフック（Hugo render hooks） - alt: ![]() の場合でもファイル名から自動生成し、alt 欠落を防ぐ - loading="lazy" / decoding="async" をデフォルト付与 - ページリソース / アセット内の画像は Hugo Image Processing で ?width=&height=&quality= 指定によりリサイズ - 例: ![説明](image.png?width=800 "title") */}} {{- $url := .Destination -}} {{- $alt := .Text | default (path.BaseName $url) -}} {{- $title := .Title -}} {{- /* サイト内バンドル画像を検索: page resources -> global resources */ -}} {{- $clean := ($url | safeURL) -}} {{- if in $clean "?" -}}{{- $clean = (strings.TrimPrefix "?" (index (split $clean "?") 0)) -}}{{- end -}} {{- $img := false -}} {{- with $.Page.Resources.GetMatch $clean -}} {{- $img = . -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $clean) -}} {{- $img = . -}} {{- end -}} {{- end -}} {{- if $img -}} {{- $opts := newScratch -}} {{- $q := "" -}} {{- if in $url "?" -}}{{- $q = (index (split $url "?") 1) -}}{{- end -}} {{- range $kv := split $q "&" -}} {{- if and (in $kv "=") (not ($opts.Get (index (split $kv "=") 0))) -}} {{- $opts.Set (index (split $kv "=") 0) (index (split $kv "=") 1) -}} {{- end -}} {{- end -}} {{- $processOpts := dict -}} {{- with $opts.Get "width" }}{{- $processOpts = merge $processOpts (dict "width" (. | int)) -}}{{- end -}} {{- with $opts.Get "height" }}{{- $processOpts = merge $processOpts (dict "height" (. | int)) -}}{{- end -}} {{- with $opts.Get "quality" }}{{- $processOpts = merge $processOpts (dict "quality" (. | int)) -}}{{- end -}} {{- $resized := $img.Process $processOpts -}} {{- else -}} {{- end -}}
